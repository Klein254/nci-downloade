    enum QState { Pending, Active, Done, Skipped, Failed, Canceled }

    class CollectionInfo
    {
        public string Name = "", Title = "", CancerTypes = "", Locations = "", Subjects = "", Species = "", Link = "", Abstract = "", Doi = "";
        public List<string> Modalities = new List<string>();
        public bool ModalitiesLoaded;
        string[] words = new string[0];
        string nameLower = "";

        public string TypeText { get { return CancerTypes != "" ? CancerTypes : Locations; } }
        public string ModText { get { return ModalitiesLoaded ? string.Join(", ", Modalities) : "…"; } }
        public int SubjectsNum { get { int n; return int.TryParse(Subjects.Replace(",", "").Trim(), out n) ? n : -1; } }
        public string SubjectsText { get { int n = SubjectsNum; return n >= 0 ? n.ToString("N0") : Subjects; } }

        public void UpdateSearch()
        {
            string all = Name + " " + Title + " " + CancerTypes + " " + Locations + " " + Species + " " + string.Join(" ", Modalities);
            words = Regex.Split(all.ToLowerInvariant(), "[^a-z0-9]+").Where(w => w != "").Distinct().ToArray();
            nameLower = Name.ToLowerInvariant();
        }

        public bool Matches(string[] terms, string raw)
        {
            if (raw != "" && nameLower.Contains(raw)) return true;
            foreach (var t in terms)
            {
                bool ok = false;
                foreach (var w in words) if (w.StartsWith(t, StringComparison.Ordinal)) { ok = true; break; }
                if (!ok) return false;
            }
            return true;
        }
    }

    class SeriesInfo
    {
        public string SeriesUID = "", StudyUID = "", Collection = "", PatientID = "", Modality = "", SeriesDesc = "",
            StudyDesc = "", StudyDate = "", SeriesDate = "", BodyPart = "", SeriesNumber = "";
        public int ImageCount;
        public long FileSize;

        public static SeriesInfo From(Dictionary<string, object> d)
        {
            var s = new SeriesInfo();
            s.SeriesUID = J.S(d, "SeriesInstanceUID");
            s.StudyUID = J.S(d, "StudyInstanceUID");
            s.Collection = J.S(d, "Collection");
            s.PatientID = J.S(d, "PatientID");
            s.Modality = J.S(d, "Modality");
            s.SeriesDesc = J.S(d, "SeriesDescription");
            if (s.SeriesDesc == "") s.SeriesDesc = J.S(d, "ProtocolName");
            s.StudyDesc = J.S(d, "StudyDesc");
            if (s.StudyDesc == "") s.StudyDesc = J.S(d, "StudyDescription");
            s.StudyDate = Util.DateOnly(J.S(d, "StudyDate"));
            s.SeriesDate = Util.DateOnly(J.S(d, "SeriesDate"));
            s.BodyPart = J.S(d, "BodyPartExamined");
            s.SeriesNumber = Util.CleanNumber(J.S(d, "SeriesNumber"));
            s.ImageCount = (int)J.L(d, "ImageCount");
            s.FileSize = J.L(d, "FileSize");
            return s;
        }

        public static int Compare(SeriesInfo a, SeriesInfo b)
        {
            int r = string.CompareOrdinal(a.StudyDate, b.StudyDate); if (r != 0) return r;
            r = string.CompareOrdinal(a.StudyUID, b.StudyUID); if (r != 0) return r;
            double x, y;
            bool hx = double.TryParse(a.SeriesNumber, NumberStyles.Float, CultureInfo.InvariantCulture, out x);
            bool hy = double.TryParse(b.SeriesNumber, NumberStyles.Float, CultureInfo.InvariantCulture, out y);
            if (hx && hy && x != y) return x.CompareTo(y);
            if (hx != hy) return hx ? -1 : 1;
            return string.CompareOrdinal(a.Modality + a.SeriesDesc, b.Modality + b.SeriesDesc);
        }
    }

    // ------------------------------------------------------------------ JSON helpers

    static class J
    {
        public static string S(Dictionary<string, object> d, string k)
        {
            object v;
            if (d == null || !d.TryGetValue(k, out v) || v == null) return "";
            if (v is string) return ((string)v).Trim();
            var dict = v as Dictionary<string, object>;
            if (dict != null) return S(dict, "rendered");
            if (v is IEnumerable)
            {
                var parts = new List<string>();
                foreach (var o in (IEnumerable)v)
                {
                    if (o == null) continue;
                    string s = WebUtility.HtmlDecode(Convert.ToString(o, CultureInfo.InvariantCulture)).Trim();
                    if (s != "" && !parts.Contains(s)) parts.Add(s);
                }
                return string.Join(", ", parts);
            }
            return Convert.ToString(v, CultureInfo.InvariantCulture);
        }

        public static long L(Dictionary<string, object> d, string k)
        {
            double x;
            return double.TryParse(S(d, k), NumberStyles.Float, CultureInfo.InvariantCulture, out x) ? (long)x : 0;
        }

        public static List<Dictionary<string, object>> ParseList(string s)
        {
            var list = new List<Dictionary<string, object>>();
            if (string.IsNullOrWhiteSpace(s)) return list;
            s = s.Trim();
            if (!s.StartsWith("[") && !s.StartsWith("{")) return list;
            var ser = new JavaScriptSerializer();
            ser.MaxJsonLength = int.MaxValue;
            ser.RecursionLimit = 256;
            object o = ser.DeserializeObject(s);
            var one = o as Dictionary<string, object>;
            if (one != null) { list.Add(one); return list; }
            var arr = o as IEnumerable;
            if (arr != null)
                foreach (var x in arr) { var d = x as Dictionary<string, object>; if (d != null) list.Add(d); }
            return list;
        }
    }

    // ------------------------------------------------------------------ TCIA web services

    static class Tcia
    {
        public const string Base = "https://services.cancerimagingarchive.net/nbia-api/services/v1/";
        public const string Web = "https://www.cancerimagingarchive.net/api/v1/collections/";
        static readonly HttpClient http;

        static Tcia()
        {
            var h = new HttpClientHandler();
            h.AutomaticDecompression = DecompressionMethods.GZip | DecompressionMethods.Deflate;
            http = new HttpClient(h);
            http.Timeout = Timeout.InfiniteTimeSpan;
            http.DefaultRequestHeaders.TryAddWithoutValidation("User-Agent", "Mozilla/5.0 (Windows NT 10.0; Win64; x64) TCIA-Downloader/1.0");
        }

        public static HttpClient Http { get { return http; } }

        public static async Task<string> GetString(string url, CancellationToken ct, int timeoutSec)
        {
            for (int attempt = 1; ; attempt++)
            {
                string err = null;
                bool retry = true;
                using (var cts = CancellationTokenSource.CreateLinkedTokenSource(ct))
                {
                    cts.CancelAfter(TimeSpan.FromSeconds(timeoutSec));
                    try
                    {
                        using (var resp = await http.GetAsync(url, cts.Token))
                        {
                            if (resp.IsSuccessStatusCode) return await resp.Content.ReadAsStringAsync();
                            int code = (int)resp.StatusCode;
                            err = "the server answered " + code + " " + resp.ReasonPhrase;
                            retry = code >= 500 || code == 429;
                        }
                    }
                    catch (OperationCanceledException) { if (ct.IsCancellationRequested) throw; err = "the request timed out"; }
                    catch (HttpRequestException ex) { err = Util.Inner(ex); }
                }
                if (!retry || attempt >= 3) throw new HttpRequestException(err);
                await Task.Delay(1000 * attempt, ct);
            }
        }

        public static async Task<List<Dictionary<string, object>>> GetList(string pathAndQuery, CancellationToken ct)
        {
            return J.ParseList(await GetString(Base + pathAndQuery, ct, 120));
        }

        // Collection descriptions (cancer type, site, subjects, abstract) from the TCIA website.
        public static async Task<List<Dictionary<string, object>>> GetWebCollections(CancellationToken ct)
        {
            const string fields = "slug,link,title,cancer_types,cancer_locations,subjects,collection_title,collection_short_title,species,collection_doi," +
                "collection_abstract,collection_summary,collection_introduction";
            Func<int, string> url = p => Web + "?per_page=100&page=" + p + "&_fields=" + fields;
            var all = new List<Dictionary<string, object>>();
            int pages = 5;
            using (var cts = CancellationTokenSource.CreateLinkedTokenSource(ct))
            {
                cts.CancelAfter(TimeSpan.FromSeconds(90));
                using (var resp = await http.GetAsync(url(1), cts.Token))
                {
                    resp.EnsureSuccessStatusCode();
                    IEnumerable<string> vals;
                    int n;
                    if (resp.Headers.TryGetValues("X-WP-TotalPages", out vals) && int.TryParse(vals.FirstOrDefault(), out n)) pages = n;
                    all.AddRange(J.ParseList(await resp.Content.ReadAsStringAsync()));
                }
            }
            if (pages > 1)
            {
                var tasks = Enumerable.Range(2, Math.Min(pages, 30) - 1).Select(p => GetString(url(p), ct, 90)).ToList();
                foreach (var t in tasks)
                {
                    try { all.AddRange(J.ParseList(await t)); }
                    catch (HttpRequestException) { }
                }
            }
            return all;
        }
    }

    // ------------------------------------------------------------------ downloading

    static class Downloader
    {
        public static long TotalReceived;
        public static double ZipRatio = 0.62;   // learned: zip bytes / uncompressed bytes
        static int ratioSamples;
        static readonly object ratioLock = new object();
        public const string StDownloading = "Downloading";
        public const string StUnpacking = "Unpacking…";
        static readonly TimeSpan Stall = TimeSpan.FromSeconds(120);

        public static string CollectionDir(string root, SeriesInfo s)
        {
            return Path.Combine(root, Util.Clean(s.Collection, 48, "Unknown collection"));
        }

        // <root>\<Collection>\<Patient>\<StudyDate StudyDesc (uid)>\<No Modality SeriesDesc (uid)>\1-001.dcm
        public static string SeriesFolder(string root, SeriesInfo s)
        {
            string coll = CollectionDir(root, s);
            string pat = Util.Clean(s.PatientID, 48, "Unknown patient");
            string study = Util.Clean(Util.JoinNonEmpty(" ", s.StudyDate, s.StudyDesc), 40, "Study") + " (" + Util.Tail(s.StudyUID, 5) + ")";
            string ser = Util.Clean(Util.JoinNonEmpty(" ", s.SeriesNumber, s.Modality, s.SeriesDesc), 44, "Series") + " (" + Util.Tail(s.SeriesUID, 5) + ")";
            string p = Path.Combine(coll, pat, study, ser);
            if (p.Length > 205)
            {
                study = Util.Clean(s.StudyDate, 12, "Study") + " (" + Util.Tail(s.StudyUID, 5) + ")";
                ser = Util.Clean(Util.JoinNonEmpty(" ", s.SeriesNumber, s.Modality), 16, "Series") + " (" + Util.Tail(s.SeriesUID, 5) + ")";
                p = Path.Combine(coll, pat, study, ser);
            }
            return p;
        }

        public static async Task DownloadOne(QueueItem q, CancellationToken ct)
        {
            SeriesInfo s = q.S;
            if (Util.IsComplete(q.Folder, s.ImageCount))
            {
                q.Status = "Already on your computer";
                q.State = QState.Skipped;
                return;
            }

            string partial = Path.Combine(q.Root, ".partial");
            Directory.CreateDirectory(partial);
            try { File.SetAttributes(partial, File.GetAttributes(partial) | FileAttributes.Hidden); } catch (IOException) { }

            try
            {
                var drive = new DriveInfo(Path.GetPathRoot(Path.GetFullPath(q.Root)));
                if (s.FileSize > 0 && drive.AvailableFreeSpace < s.FileSize * 2 + 200L * 1024 * 1024)
                    throw new PermanentException("Not enough free disk space on " + drive.Name);
            }
            catch (ArgumentException) { }

            string zipPath = Path.Combine(partial, Util.Clean(s.SeriesUID, 80, "series") + ".zip");
            string url = Tcia.Base + "getImage?NewFileNames=Yes&SeriesInstanceUID=" + Uri.EscapeDataString(s.SeriesUID);
            const int MaxAttempts = 4;
            for (int attempt = 1; ; attempt++)
            {
                ct.ThrowIfCancellationRequested();
                Interlocked.Exchange(ref q.Received, 0);
                q.Status = attempt == 1 ? "Connecting…" : "Retrying (" + attempt + " of " + MaxAttempts + ")…";
                string err = null;
                try { await FetchToFile(url, zipPath, q, ct); }
                catch (OperationCanceledException) { if (ct.IsCancellationRequested) throw; err = "the connection stalled"; }
                catch (HttpRequestException ex) { if (ct.IsCancellationRequested) throw new OperationCanceledException(ct); err = Util.Inner(ex); }
                catch (IOException ex) { if (ct.IsCancellationRequested) throw new OperationCanceledException(ct); err = ex.Message; }
                catch (WebException ex) { if (ct.IsCancellationRequested) throw new OperationCanceledException(ct); err = ex.Message; }
                catch (ObjectDisposedException) { if (ct.IsCancellationRequested) throw new OperationCanceledException(ct); err = "the connection stalled"; }
                if (err == null) break;
                Util.TryDelete(zipPath);
                if (attempt >= MaxAttempts) throw new IOException("Download failed: " + err);
                await Task.Delay(TimeSpan.FromSeconds(3 * attempt), ct);
            }

            q.ZipBytes = new FileInfo(zipPath).Length;
            if (q.ZipBytes < 22)
            {
                Util.TryDelete(zipPath);
                throw new PermanentException("TCIA sent no data for this series (it may be limited-access)");
            }

            q.Status = StUnpacking;
            string tmp = q.Folder + ".part";
            int files = 0;
            try
            {
                Util.TryDeleteDir(tmp);
                Directory.CreateDirectory(tmp);
                using (var za = ZipFile.OpenRead(zipPath))
                {
                    foreach (var e in za.Entries)
                    {
                        ct.ThrowIfCancellationRequested();
                        string name = Path.GetFileName(e.FullName.Replace('/', '\\'));
                        if (string.IsNullOrEmpty(name)) continue;
                        if (name.Equals("LICENSE", StringComparison.OrdinalIgnoreCase) || name.StartsWith("LICENSE.", StringComparison.OrdinalIgnoreCase))
                        {
                            string lic = Path.Combine(q.CollectionDir, "LICENSE.txt");
                            try { if (!File.Exists(lic)) e.ExtractToFile(lic, false); } catch (IOException) { }
                            continue;
                        }
                        e.ExtractToFile(Path.Combine(tmp, name), true);
                        files++;
                    }
                }
                if (files == 0) throw new PermanentException("The download contained no DICOM files");
                Util.TryDeleteDir(q.Folder);
                Directory.Move(tmp, q.Folder);
            }
            catch (InvalidDataException)
            {
                Util.TryDeleteDir(tmp);
                throw new IOException("The downloaded file was damaged - please retry");
            }
            catch
            {
                Util.TryDeleteDir(tmp);
                throw;
            }
            finally
            {
                Util.TryDelete(zipPath);
            }

            if (s.FileSize > 0)
            {
                double r = (double)q.ZipBytes / s.FileSize;
                lock (ratioLock)
                {
                    if (r > 0.05 && r < 1.5)
                    {
                        ZipRatio = (ZipRatio * (ratioSamples + 1) + r) / (ratioSamples + 2);
                        ratioSamples = Math.Min(ratioSamples + 1, 20);
                    }
                }
            }
            q.Files = files;
            q.Status = "Done · " + files.ToString("N0") + (files == 1 ? " file" : " files");
            q.State = QState.Done;
        }

        static async Task FetchToFile(string url, string path, QueueItem q, CancellationToken ct)
        {
            using (var stall = CancellationTokenSource.CreateLinkedTokenSource(ct))
            {
                stall.CancelAfter(Stall);
                using (var resp = await Tcia.Http.GetAsync(url, HttpCompletionOption.ResponseHeadersRead, stall.Token))
                using (stall.Token.Register(() => { try { resp.Dispose(); } catch (Exception) { } }))
                {
                    int code = (int)resp.StatusCode;
                    if (code >= 400 && code < 500 && code != 408 && code != 429)
                        throw new PermanentException("TCIA refused this series (" + code + "); it may be limited-access");
                    if (!resp.IsSuccessStatusCode) throw new HttpRequestException("the server answered " + code + " " + resp.ReasonPhrase);
                    q.Status = StDownloading;
                    using (var src = await resp.Content.ReadAsStreamAsync())
                    using (var dst = new FileStream(path, FileMode.Create, FileAccess.Write, FileShare.None, 1 << 16, true))
                    {
                        var buf = new byte[1 << 17];
                        while (true)
                        {
                            int n = await src.ReadAsync(buf, 0, buf.Length, stall.Token);
                            if (n <= 0) break;
                            await dst.WriteAsync(buf, 0, n);
                            Interlocked.Add(ref q.Received, n);
                            Interlocked.Add(ref TotalReceived, n);
                            stall.CancelAfter(Stall);
                        }
                    }
                }
            }
        }
    }

    // ------------------------------------------------------------------ utilities

    static class Util
    {
        static readonly char[] Bad = Path.GetInvalidFileNameChars();
        static readonly HashSet<string> Reserved = new HashSet<string>(StringComparer.OrdinalIgnoreCase) {
            "CON", "PRN", "AUX", "NUL", "COM1", "COM2", "COM3", "COM4", "COM5", "COM6", "COM7", "COM8", "COM9",
            "LPT1", "LPT2", "LPT3", "LPT4", "LPT5", "LPT6", "LPT7", "LPT8", "LPT9" };
        static readonly string[] ModOrder = { "CT", "MR", "PT", "NM", "CR", "DX", "MG", "US", "XA", "RF", "RTSTRUCT", "RTPLAN", "RTDOSE", "RTIMAGE", "RTRECORD", "SEG", "REG", "SR", "PR", "KO", "SM", "OT" };

        public static string Clean(string s, int max, string fallback)
        {
            var sb = new StringBuilder();
            foreach (char ch in (s ?? "")) sb.Append(Array.IndexOf(Bad, ch) >= 0 || char.IsControl(ch) ? '_' : ch);
            string r = Regex.Replace(sb.ToString(), @"\s+", " ").Trim().Trim('.', ' ');
            if (r.Length > max) r = r.Substring(0, max).Trim().Trim('.', ' ');
            if (r == "") r = fallback;
            if (Reserved.Contains(r)) r = "_" + r;
            return r;
        }

        public static string Tail(string s, int n) { s = s ?? ""; return s.Length <= n ? s : s.Substring(s.Length - n); }

        public static string JoinNonEmpty(string sep, params string[] parts)
        {
            return string.Join(sep, parts.Where(p => !string.IsNullOrWhiteSpace(p)).Select(p => p.Trim()));
        }

        public static string Inner(Exception ex)
        {
            while (ex.InnerException != null) ex = ex.InnerException;
            return ex.Message;
        }

        public static string DateOnly(string s)
        {
            if (string.IsNullOrEmpty(s)) return "";
            var m = Regex.Match(s, @"^\d{4}-\d{2}-\d{2}");
            if (m.Success) return m.Value;
            m = Regex.Match(s, @"^(\d{2})-(\d{2})-(\d{4})");
            if (m.Success) return m.Groups[3].Value + "-" + m.Groups[1].Value + "-" + m.Groups[2].Value;
            return s.Length > 10 ? s.Substring(0, 10) : s;
        }

        public static string CleanNumber(string s)
        {
            double d;
            if (double.TryParse(s, NumberStyles.Float, CultureInfo.InvariantCulture, out d) && Math.Abs(d - Math.Round(d)) < 1e-9)
                return ((long)Math.Round(d)).ToString(CultureInfo.InvariantCulture);
            return s ?? "";
        }

        public static string Size(long b)
        {
            if (b <= 0) return "";
            string[] u = { "B", "KB", "MB", "GB", "TB" };
            double v = b; int i = 0;
            while (v >= 1024 && i < u.Length - 1) { v /= 1024; i++; }
            string num = i == 0 ? v.ToString("0") : v < 10 ? v.ToString("0.0") : v.ToString("0");
            return num + " " + u[i];
        }

        public static string Duration(double sec)
        {
            if (double.IsNaN(sec) || double.IsInfinity(sec) || sec < 0) return "";
            if (sec < 60) return "less than a minute";
            if (sec < 3600) return (int)Math.Ceiling(sec / 60) + " min";
            double h = sec / 3600;
            return h < 10 ? h.ToString("0.0") + " hours" : (int)h + " hours";
        }

        public static string ShortPath(string p, int max)
        {
            if (p == null || p.Length <= max) return p;
            string root = Path.GetPathRoot(p) ?? "";
            var parts = p.Substring(root.Length).TrimEnd('\\').Split('\\');
            string tail = parts[parts.Length - 1];
            for (int i = parts.Length - 2; i >= 0; i--)
            {
                string cand = parts[i] + "\\" + tail;
                if (root.Length + 2 + cand.Length > max) break;
                tail = cand;
            }
            return root + "…\\" + tail;
        }

        public static string SeriesLabel(SeriesInfo s) { return s.SeriesDesc != "" ? s.SeriesDesc : "(no description)"; }

        public static string Norm(string s) { return Regex.Replace((s ?? "").ToLowerInvariant(), "[^a-z0-9]", ""); }

        public static string StripHtml(string html)
        {
            if (string.IsNullOrEmpty(html)) return "";
            string s = Regex.Replace(html, @"<\s*(br|/p|/li|/h\d|/div)[^>]*>", "\n", RegexOptions.IgnoreCase);
            s = Regex.Replace(s, @"<\s*li[^>]*>", "\n• ", RegexOptions.IgnoreCase);
            s = Regex.Replace(s, "<[^>]+>", "");
            s = WebUtility.HtmlDecode(s).Replace(' ', ' ');
            s = Regex.Replace(s, @"[ \t\r\f\v]+", " ");
            s = Regex.Replace(s, @" *\n *", "\n");
            s = Regex.Replace(s, @"\n{2,}", "\n\n").Trim();
            return s.Replace("\n", "\r\n");
        }

        public static bool IsComplete(string folder, int count)
        {
            try { return Directory.Exists(folder) && Directory.GetFiles(folder).Length >= Math.Max(1, count); }
            catch (Exception) { return false; }
        }

        public static void TryDelete(string f) { try { if (File.Exists(f)) File.Delete(f); } catch (Exception) { } }
        public static void TryDeleteDir(string d) { try { if (Directory.Exists(d)) Directory.Delete(d, true); } catch (Exception) { } }

        public static List<string> SortModalities(IEnumerable<string> mods)
        {
            return mods.Distinct().OrderBy(m => { int i = Array.IndexOf(ModOrder, m); return i < 0 ? 100 : i; })
                .ThenBy(m => m, StringComparer.Ordinal).ToList();
        }

        public static string ModalityLabel(string m)
        {
            switch (m)
            {
                case "MR": return "MR (MRI)";
                case "PT": return "PT (PET)";
                case "NM": return "NM (nuclear medicine)";
                case "CR": return "CR (X-ray)";
                case "DX": return "DX (X-ray)";
                case "MG": return "MG (mammography)";
                case "US": return "US (ultrasound)";
                case "XA": return "XA (angiography)";
                case "RTSTRUCT": return "RTSTRUCT (contours)";
                case "RTPLAN": return "RTPLAN (treatment plan)";

        // A .tcia manifest is a few key=value lines followed by one SeriesInstanceUID per line.
        public static List<string> ParseManifest(IEnumerable<string> lines)
        {
            return lines.Select(l => l.Trim()).Where(l => Regex.IsMatch(l, @"^[0-9]+(\.[0-9]+){2,}$")).Distinct().ToList();
        }

    static class AppIcon
    {
        public static Bitmap Render(int size)
        {
            var bmp = new Bitmap(size, size, PixelFormat.Format32bppArgb);
            using (var g = Graphics.FromImage(bmp))
            {
                g.SmoothingMode = SmoothingMode.AntiAlias;
                g.PixelOffsetMode = PixelOffsetMode.HighQuality;
                g.Clear(Color.Transparent);
                float s = size, pad = Math.Max(0.5f, s * 0.04f);
                var r = new RectangleF(pad, pad, s - 2 * pad, s - 2 * pad);
                using (var path = Rounded(r, s * 0.22f))
                using (var br = new LinearGradientBrush(new PointF(0, 0), new PointF(s, s), Color.FromArgb(26, 164, 170), Color.FromArgb(13, 50, 80)))
                    g.FillPath(br, path);
                float rw = Math.Max(1.4f, s * 0.085f);
                using (var pen = new Pen(Color.FromArgb(110, 255, 255, 255), rw))
                    g.DrawEllipse(pen, s * 0.20f, s * 0.20f, s * 0.60f, s * 0.60f);
                using (var b = new SolidBrush(Color.White))
                {
                    float cx = s * 0.5f, w = Math.Max(1.5f, s * 0.12f);
                    g.FillRectangle(b, cx - w / 2, s * 0.29f, w, s * 0.24f);
                    g.FillPolygon(b, new PointF[] { new PointF(s * 0.32f, s * 0.49f), new PointF(s * 0.68f, s * 0.49f), new PointF(cx, s * 0.72f) });
                }
            }
            return bmp;
        }

        static GraphicsPath Rounded(RectangleF r, float rad)
        {
            var p = new GraphicsPath();
            float d = rad * 2;
            p.AddArc(r.X, r.Y, d, d, 180, 90);
            p.AddArc(r.Right - d, r.Y, d, d, 270, 90);
            p.AddArc(r.Right - d, r.Bottom - d, d, d, 0, 90);
            p.AddArc(r.X, r.Bottom - d, d, d, 90, 90);
            p.CloseFigure();
            return p;
        }

        public static Icon MakeIcon() { return new Icon(new MemoryStream(BuildIco(new int[] { 16, 20, 24, 32, 40, 48, 64, 256 }))); }

        public static byte[] BuildIco(int[] sizes)
        {
            var images = new List<byte[]>();
            foreach (int sz in sizes) using (var bmp = Render(sz)) images.Add(sz >= 256 ? Png(bmp) : Dib(bmp));
            var ms = new MemoryStream();
            var w = new BinaryWriter(ms);
            w.Write((short)0); w.Write((short)1); w.Write((short)sizes.Length);
            int offset = 6 + 16 * sizes.Length;
            for (int i = 0; i < sizes.Length; i++)
            {
                byte dim = (byte)(sizes[i] >= 256 ? 0 : sizes[i]);
                w.Write(dim); w.Write(dim); w.Write((byte)0); w.Write((byte)0);
                w.Write((short)1); w.Write((short)32); w.Write(images[i].Length); w.Write(offset);
                offset += images[i].Length;
            }
            foreach (var img in images) w.Write(img);
            w.Flush();
            return ms.ToArray();
        }

        static byte[] Png(Bitmap b) { using (var ms = new MemoryStream()) { b.Save(ms, ImageFormat.Png); return ms.ToArray(); } }

        static byte[] Dib(Bitmap b)
        {
            int n = b.Width;
            var ms = new MemoryStream();
            var w = new BinaryWriter(ms);
            w.Write(40); w.Write(n); w.Write(n * 2); w.Write((short)1); w.Write((short)32);
            w.Write(0); w.Write(n * n * 4); w.Write(0); w.Write(0); w.Write(0); w.Write(0);
            for (int y = n - 1; y >= 0; y--)
                for (int x = 0; x < n; x++) { Color c = b.GetPixel(x, y); w.Write(c.B); w.Write(c.G); w.Write(c.R); w.Write(c.A); }
            w.Write(new byte[((n + 31) / 32) * 4 * n]);
            w.Flush();
            return ms.ToArray();
        }
    }

    // WinForms only raises Click if the window under the cursor at mouse-up is the button itself, so
    // any overlay window (screen tools, accessibility aids) silently swallows clicks. Fall back to the
    // mouse-up position when that happens.
    class FlatButton : Button
    {
        bool pressed, clickedByBase;
        protected override void OnMouseDown(MouseEventArgs e)
        {
            if (e.Button == MouseButtons.Left) pressed = true;
            base.OnMouseDown(e);
        }
        protected override void OnMouseUp(MouseEventArgs e)
        {
            bool fire = e.Button == MouseButtons.Left && pressed && Enabled && ClientRectangle.Contains(e.Location);
            pressed = false;
            clickedByBase = false;
            base.OnMouseUp(e);
            if (fire && !clickedByBase && !IsDisposed) OnClick(EventArgs.Empty);
        }
        protected override void OnClick(EventArgs e)
        {
            clickedByBase = true;
            base.OnClick(e);
        }
    }

    class ThinBar : Control

        }
    }

        void AutoFit(ListView lv, int fillCol, int minW)
        {
            if (lv.Columns.Count <= fillCol) return;
            int other = 0;
            for (int i = 0; i < lv.Columns.Count; i++) if (i != fillCol) other += lv.Columns[i].Width;
            int w = lv.ClientSize.Width - other - 1;
            if (!Native.HasVScroll(lv)) w -= SystemInformation.VerticalScrollBarWidth;   // leave room for it to appear
            lv.Columns[fillCol].Width = Math.Max(Px(minW), w);
        }

        string CacheFile { get { return Path.Combine(appDir, "catalog.json"); } }
        const string CacheVersion = "2";

        bool LoadCache(out DateTime savedUtc)
        {
            savedUtc = DateTime.MinValue;
            try
            {
                if (!File.Exists(CacheFile)) return false;
                var ser = new JavaScriptSerializer(); ser.MaxJsonLength = int.MaxValue;
                var root = ser.DeserializeObject(File.ReadAllText(CacheFile, Encoding.UTF8)) as Dictionary<string, object>;
                if (root == null || !root.ContainsKey("collections") || J.S(root, "v") != CacheVersion) return false;
                DateTime.TryParse(J.S(root, "saved"), CultureInfo.InvariantCulture, DateTimeStyles.RoundtripKind, out savedUtc);
                webLoaded = J.S(root, "web") == "True";
                var list = new List<CollectionInfo>();
                foreach (var o in (IEnumerable)root["collections"])
                {
                    var d = o as Dictionary<string, object>;
                    if (d == null) continue;
                    var c = new CollectionInfo();
                    c.Name = J.S(d, "name");
                    if (c.Name == "") continue;
                    c.Title = J.S(d, "title"); c.CancerTypes = J.S(d, "cancer"); c.Locations = J.S(d, "loc");
                    c.Subjects = J.S(d, "subjects"); c.Species = J.S(d, "species"); c.Link = J.S(d, "link");
                    c.Abstract = J.S(d, "abstract"); c.Doi = J.S(d, "doi");
                    object mods;
                    if (d.TryGetValue("mods", out mods) && mods is IEnumerable && !(mods is string))
                    {
                        c.Modalities = ((IEnumerable)mods).Cast<object>().Select(x => Convert.ToString(x)).Where(x => !string.IsNullOrEmpty(x)).ToList();
                        c.ModalitiesLoaded = true;
                    }
                    c.UpdateSearch();
                    list.Add(c);
                }
                if (list.Count == 0) return false;
                collections = list;
                return true;
            }
            catch (Exception) { return false; }
        }

        void SaveCache()
        {
            try
            {
                var list = collections.Select(c => (object)new Dictionary<string, object> {
                    { "name", c.Name }, { "title", c.Title }, { "cancer", c.CancerTypes }, { "loc", c.Locations },
                    { "subjects", c.Subjects }, { "species", c.Species }, { "link", c.Link }, { "abstract", c.Abstract },
                    { "doi", c.Doi }, { "mods", c.ModalitiesLoaded ? (object)c.Modalities.ToArray() : null } }).ToArray();
                var root = new Dictionary<string, object> { { "v", CacheVersion }, { "saved", DateTime.UtcNow.ToString("o") }, { "web", webLoaded }, { "collections", list } };
                var ser = new JavaScriptSerializer(); ser.MaxJsonLength = int.MaxValue;
                Directory.CreateDirectory(appDir);
                string tmp = CacheFile + ".tmp";
                File.WriteAllText(tmp, ser.Serialize(root), Encoding.UTF8);
                if (File.Exists(CacheFile)) File.Delete(CacheFile);
                File.Move(tmp, CacheFile);
            }
            catch (Exception) { }
        }

        async Task LoadCatalog(bool force)
        {
            if (catalogBusy) return;
            catalogBusy = true;
            btnRefresh.Enabled = false;
            try
            {
                DateTime saved;
                if (!force && collections.Count == 0 && LoadCache(out saved))
                {
                    BuildModalityCombo();
                    ApplyCollectionFilter();
                    bool fresh = (DateTime.UtcNow - saved).TotalDays < 7 && webLoaded && collections.All(c => c.ModalitiesLoaded);
                    SetStatus("Ready · " + collections.Count + " public collections.");
                    if (fresh) return;
                }

                SetStatus(collections.Count > 0 ? "Checking TCIA for catalog updates…" : "Connecting to The Cancer Imaging Archive…");
                lblCollInfo.Text = collections.Count > 0 ? lblCollInfo.Text : "Loading the list of collections…";
                var tWeb = Tcia.GetWebCollections(CancellationToken.None);
                List<Dictionary<string, object>> nb;
                try { nb = await Tcia.GetList("getCollectionValues", CancellationToken.None); }
                catch (Exception ex)
                {
                    SetStatus("Couldn't reach TCIA (" + Util.Inner(ex) + "). Check your internet connection, then click Refresh catalog.", true);
                    if (collections.Count == 0) lblCollInfo.Text = "Not connected to TCIA";
                    return;
                }

                var old = new Dictionary<string, CollectionInfo>(StringComparer.OrdinalIgnoreCase);
                foreach (var c in collections) old[c.Name] = c;
                var seen = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
                var list = new List<CollectionInfo>();
                foreach (var r in nb)
                {
                    string n = J.S(r, "Collection");
                    if (n == "" || !seen.Add(n)) continue;
                    CollectionInfo c;
                    if (!old.TryGetValue(n, out c)) { c = new CollectionInfo(); c.Name = n; }
                    if (force) c.ModalitiesLoaded = false;
                    c.UpdateSearch();
                    list.Add(c);
                }
                collections = list;
                ApplyCollectionFilter();

                try { MergeWeb(await tWeb); webLoaded = true; }
                catch (Exception) { /* descriptions are optional */ }
                ApplyCollectionFilter();
                if (curCollection != null) ShowCollectionDetails(curCollection);
                SaveCache();

                await IndexModalities();
                SaveCache();
                BuildModalityCombo();
                ApplyCollectionFilter();
                SetStatus("Ready · " + collections.Count + " public collections.");
            }
            finally
            {
                catalogBusy = false;
                btnRefresh.Enabled = true;
            }
        }

        void MergeWeb(List<Dictionary<string, object>> web)
        {
            var map = new Dictionary<string, Dictionary<string, object>>();
            foreach (var r in web)
                foreach (var key in new[] { J.S(r, "collection_short_title"), J.S(r, "title"), J.S(r, "slug") })
                {
                    string k = Util.Norm(WebUtility.HtmlDecode(key));
                    if (k != "" && !map.ContainsKey(k)) map[k] = r;
                }
            foreach (var c in collections)
            {
                Dictionary<string, object> r;
                if (!map.TryGetValue(Util.Norm(c.Name), out r)) continue;
                string title = WebUtility.HtmlDecode(J.S(r, "collection_title"));
                if (title == "") title = WebUtility.HtmlDecode(J.S(r, "title"));
                c.Title = title;
                c.CancerTypes = J.S(r, "cancer_types");
                c.Locations = J.S(r, "cancer_locations");
                c.Subjects = J.S(r, "subjects");
                c.Species = J.S(r, "species");
                c.Link = J.S(r, "link");
                c.Doi = J.S(r, "collection_doi");
                string text = "";
                foreach (var f in new[] { "collection_abstract", "collection_summary", "collection_introduction" })
                {
                    text = Util.StripHtml(J.S(r, f));
                    if (text != "") break;
                }
                c.Abstract = text;
                c.UpdateSearch();
            }
        }

        async Task IndexModalities()
        {
            var todo = collections.Where(c => !c.ModalitiesLoaded).ToList();
            if (todo.Count == 0) return;
            indexing = true; indexDone = 0; indexTotal = todo.Count;
            UpdateCollInfoLabel();
            var sem = new SemaphoreSlim(8);
            var tasks = todo.Select(async c =>
            {
                await sem.WaitAsync();
                try
                {
                    var rows = await Tcia.GetList("getModalityValues?Collection=" + Esc(c.Name), CancellationToken.None);
                    c.Modalities = Util.SortModalities(rows.Select(r => J.S(r, "Modality")).Where(m => m != ""));
                    c.ModalitiesLoaded = true;
                    c.UpdateSearch();
                    foreach (ListViewItem it in lvColl.Items)
                        if (it.Tag == c) { it.SubItems[3].Text = c.ModText; break; }
                }
                catch (Exception) { }
                finally
                {
                    sem.Release();
                    indexDone++;
                    UpdateCollInfoLabel();
                }
            }).ToList();
            await Task.WhenAll(tasks);
            indexing = false;
            UpdateCollInfoLabel();
        }

        void BuildModalityCombo()
        {
            string cur = SelectedModality();
            var counts = new Dictionary<string, int>();
            foreach (var c in collections)
                foreach (var m in c.Modalities) { int n; counts.TryGetValue(m, out n); counts[m] = n + 1; }
            suppressMod = true;
            cbModality.BeginUpdate();
            try
            {
                cbModality.Items.Clear();
                cbModality.Items.Add(new ModItem(null, "All modalities", 0));
                int sel = 0;
                foreach (var m in Util.SortModalities(counts.Keys))
                {
                    cbModality.Items.Add(new ModItem(m, Util.ModalityLabel(m), counts[m]));
                    if (m == cur) sel = cbModality.Items.Count - 1;
                }
                cbModality.SelectedIndex = sel;
            }
            finally { cbModality.EndUpdate(); suppressMod = false; }
            int w = cbModality.Width;
            foreach (var o in cbModality.Items) w = Math.Max(w, TextRenderer.MeasureText(o.ToString(), cbModality.Font).Width + Px(28));
            cbModality.DropDownWidth = w;
        }

        void ApplyCollectionFilter()
        {
            string raw = tbCollFilter.Text.Trim().ToLowerInvariant();
            string[] terms = Regex.Split(raw, "[^a-z0-9]+").Where(x => x != "").ToArray();
            string mod = SelectedModality();
            var shown = collections.Where(c => c.Matches(terms, raw) && (mod == null || c.Modalities.Contains(mod))).ToList();
            Comparison<CollectionInfo> cmp;
            switch (sortCol)
            {
                case 1: cmp = (a, b) => CompareText(a.TypeText, b.TypeText); break;
                case 2: cmp = (a, b) => a.SubjectsNum.CompareTo(b.SubjectsNum); break;
                case 3: cmp = (a, b) => CompareText(a.ModText, b.ModText); break;
                default: cmp = (a, b) => Native.StrCmpLogicalW(a.Name, b.Name); break;
            }
            shown.Sort((a, b) => { int r = cmp(a, b); if (r == 0) r = Native.StrCmpLogicalW(a.Name, b.Name); return sortAsc ? r : -r; });

            suppressColl = true;
            lvColl.BeginUpdate();
            try
            {
                lvColl.Items.Clear();
                foreach (var c in shown)
                {
                    var it = new ListViewItem(new string[] { c.Name, c.TypeText, c.SubjectsText, c.ModText });
                    it.Tag = c;
                    lvColl.Items.Add(it);
                    if (c == curCollection) { it.Selected = true; it.Focused = true; }
                }
            }
            finally { lvColl.EndUpdate(); suppressColl = false; }
            lastShownCount = shown.Count;
            UpdateCollInfoLabel();
        }

        static int CompareText(string a, string b)
        {
            if (a == "" && b != "") return 1;
            if (b == "" && a != "") return -1;
            return string.Compare(a, b, StringComparison.OrdinalIgnoreCase);
        }

        async Task LoadPatients()
        {
            var c = curCollection;
            if (c == null) return;
            string mod = SelectedModality();
            string key = c.Name + "|" + (mod ?? "*");
            int token = ++patToken;
            hdrPat.Text = mod == null ? "Patients in " + c.Name : "Patients with " + mod;
            patientsAll = new List<string>();
            ApplyPatientFilter();
            lblPatCount.Text = "Loading patients…";
            ClearSeries("Select a patient to see their scans.");

            List<string> list;
            if (!patientCache.TryGetValue(key, out list))
            {
                SetStatus("Loading patients in " + c.Name + "…");
                try
                {
                    string q = mod == null
                        ? "getPatient?Collection=" + Esc(c.Name)
                        : "getPatientByCollectionAndModality?Collection=" + Esc(c.Name) + "&Modality=" + Esc(mod);
                    var rows = await Tcia.GetList(q, CancellationToken.None);
                    list = rows.Select(r => { string id = J.S(r, "PatientId"); return id != "" ? id : J.S(r, "PatientID"); })
                        .Where(x => x != "").Distinct().ToList();
                    list.Sort((a, b) => Native.StrCmpLogicalW(a, b));
                    patientCache[key] = list;
                }
                catch (Exception ex)
                {
                    if (token == patToken)
                    {
                        lblPatCount.Text = "Couldn't load patients";
                        SetStatus("Couldn't load patients: " + Util.Inner(ex), true);
                    }
                    return;
                }
            }
            if (token != patToken) return;
            patientsAll = list;
            ApplyPatientFilter();
            SetStatus(list.Count.ToString("N0") + " patients in " + c.Name + (mod != null ? " with " + mod + " scans" : "") + ".");
            if (list.Count == 0 && mod != null) lblPatCount.Text = "No patients with " + mod + " in this collection";
            TrySelectPendingPatient();
        }

        async void DownloadSelectedPatients()
        {
            var c = curCollection;
            if (c == null) return;
            var pids = lvPat.SelectedIndices.Cast<int>().Where(i => i >= 0 && i < patientsShown.Count).Select(i => patientsShown[i]).ToList();
            if (pids.Count == 0) return;
            string mod = SelectedModality();
            if (pids.Count > 10)
            {
                var r = MessageBox.Show(this,
                    "Download " + (mod == null ? "every series" : "every " + mod + " series") + " for " + pids.Count.ToString("N0") +
                    " patients in " + c.Name + "?\n\nThis can be a lot of data. Each patient's scans are added to the download list " +
                    "as they're found, and you can cancel at any time.",
                    "Download " + pids.Count.ToString("N0") + " patients", MessageBoxButtons.YesNo, MessageBoxIcon.Question);
                if (r != DialogResult.Yes) return;
            }
            btnPatDownload.Enabled = false;
            int done = 0, added = 0, failed = 0;
            long bytes = 0;
            var sem = new SemaphoreSlim(4);
            var tasks = pids.Select(async pid =>
            {
                await sem.WaitAsync();
                try
                {
                    var list = await GetPatientSeries(c.Name, pid);
                    var sel = list.Where(s => mod == null || s.Modality == mod).ToList();
                    added += Enqueue(sel);
                    bytes += sel.Sum(s => s.FileSize);
                }
                catch (Exception) { failed++; }
                finally
                {
                    sem.Release();
                    done++;
                    if (pids.Count > 1) SetStatus("Adding patients to Downloads… " + done + " of " + pids.Count);
                }
            }).ToList();
            await Task.WhenAll(tasks);
            UpdatePatientButtons();
            if (curPatient != null && pids.Contains(curPatient) && curSeries.Count > 0) FillSeries();
            SetStatus("Added " + added + " series" + (bytes > 0 ? " (" + Util.Size(bytes) + ")" : "") + " from " + pids.Count +
                (pids.Count == 1 ? " patient" : " patients") + " to Downloads" +
                (failed > 0 ? "; " + failed + " patient(s) couldn't be read - try again" : "") + ".", failed > 0);
        }

        async Task<List<SeriesInfo>> GetPatientSeries(string coll, string pid)
        {
            string key = coll + "|" + pid;
            List<SeriesInfo> list;
            if (seriesCache.TryGetValue(key, out list)) return list;
            var tSeries = Tcia.GetList("getSeries?Collection=" + Esc(coll) + "&PatientID=" + Esc(pid), CancellationToken.None);
            var tStudies = Tcia.GetList("getPatientStudy?Collection=" + Esc(coll) + "&PatientID=" + Esc(pid), CancellationToken.None);
            var rows = await tSeries;
            var studies = new Dictionary<string, Dictionary<string, object>>();
            try
            {
                foreach (var r in await tStudies)
                {
                    string u = J.S(r, "StudyInstanceUID");
                    if (u != "") studies[u] = r;
                }
            }
            catch (Exception) { }
            list = rows.Select(SeriesInfo.From).Where(s => s.SeriesUID != "").ToList();
            foreach (var s in list)
            {
                Dictionary<string, object> st;
                if (!studies.TryGetValue(s.StudyUID, out st)) continue;
                if (s.StudyDesc == "") s.StudyDesc = J.S(st, "StudyDescription");
                if (s.StudyDate == "") s.StudyDate = Util.DateOnly(J.S(st, "StudyDate"));
            }
            list.Sort(SeriesInfo.Compare);
            seriesCache[key] = list;
            return list;
        }

        async Task LoadSeries(string pid)
        {
            var c = curCollection;
            if (c == null) return;
            int token = ++seriesToken;
            curPatient = pid;
            curSeries = new List<SeriesInfo>();
            hdrSeries.Text = "Series for " + pid;
            suppressCheck = true;
            lvSeries.Items.Clear(); lvSeries.Groups.Clear(); seriesRows.Clear();
            suppressCheck = false;
            lblSeriesSum.Text = "Loading series…";
            btnDownload.Enabled = false;
            List<SeriesInfo> list;
            try { list = await GetPatientSeries(c.Name, pid); }
            catch (Exception ex)
            {
                if (token == seriesToken) lblSeriesSum.Text = "Couldn't load series: " + Util.Inner(ex);
                return;
            }
            if (token != seriesToken) return;
            curSeries = list;
            FillSeries();
        }

        bool InQueue(SeriesInfo s)
        {
            QueueItem q;
            return byUid.TryGetValue(s.SeriesUID, out q) && (q.State == QState.Pending || q.State == QState.Active);
        }

        {
            string mod = SelectedModality();
            suppressCheck = true;
            lvSeries.BeginUpdate();
            try
            {
                lvSeries.Items.Clear(); lvSeries.Groups.Clear(); seriesRows.Clear();
                var groups = new Dictionary<string, ListViewGroup>();
                foreach (var g in curSeries.GroupBy(s => s.StudyUID))

                    var first = g.First();

                    var grp = new ListViewGroup(g.Key, head);

                    groups[g.Key] = grp;

                {
                    bool onDisk = Util.IsComplete(Downloader.SeriesFolder(saveRoot, s), s.ImageCount);

                    seriesRows[s.SeriesUID] = it;
                }

        }

        async void LoadManifest(string path)
        {
            List<string> uids;
            try { uids = Util.ParseManifest(File.ReadAllLines(path)); }
            catch (Exception ex)
            {
                MessageBox.Show(this, "Couldn't read " + Path.GetFileName(path) + ":\n\n" + ex.Message, "Open manifest", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                return;
            }
            string name = Path.GetFileName(path);
            if (uids.Count == 0)
            {
                MessageBox.Show(this, "No series were found in " + name + ".\n\nMake sure it's a .tcia manifest downloaded from the TCIA website.",
                    "Open manifest", MessageBoxButtons.OK, MessageBoxIcon.Information);
                return;
            }
            int done = 0, added = 0, missing = 0;
            SetStatus("Reading manifest " + name + ": 0 of " + uids.Count + " series…");
            var sem = new SemaphoreSlim(8);
            var tasks = uids.Select(async uid =>
            {
                await sem.WaitAsync();
                SeriesInfo s = null;
                try
                {
                    var rows = await Tcia.GetList("getSeries?SeriesInstanceUID=" + Esc(uid), CancellationToken.None);
                    if (rows.Count > 0) s = SeriesInfo.From(rows[0]);
                }
                catch (Exception) { }
                finally { sem.Release(); }
                if (s == null || s.SeriesUID == "")
                {
                    s = new SeriesInfo();
                    s.SeriesUID = uid;
                    s.Collection = "From manifest " + Path.GetFileNameWithoutExtension(name);
                    s.PatientID = "Unknown patient";
                    missing++;
                }
                added += Enqueue(new[] { s });
                done++;
                if (done % 5 == 0 || done == uids.Count) SetStatus("Reading manifest " + name + ": " + done + " of " + uids.Count + " series…");
            }).ToList();
            await Task.WhenAll(tasks);
            SetStatus("Manifest " + name + ": added " + added + " series to Downloads" +
                (missing > 0 ? " (" + missing + " weren't found in the public archive)" : "") + ".", missing > 0);
        }

        // ============================================================ download queue

        int Enqueue(IEnumerable<SeriesInfo> items)
        {
            int added = 0;
            lvQueue.BeginUpdate();
            try
            {
                foreach (var s in items)
                {
                    if (s == null || s.SeriesUID == "") continue;
                    QueueItem existing;
                    if (byUid.TryGetValue(s.SeriesUID, out existing))
                    {
                        if (existing.State == QState.Failed || existing.State == QState.Canceled)
                        {
                            existing.Status = "Waiting"; existing.Seen = false; existing.State = QState.Pending; added++;
                        }
                        continue;
                    }
                    var q = new QueueItem();
                    q.S = s;
                    q.Root = saveRoot;
                    q.CollectionDir = Downloader.CollectionDir(saveRoot, s);
                    q.Folder = Downloader.SeriesFolder(saveRoot, s);
                    q.Item = new ListViewItem(new string[] {
                        s.PatientID, s.Modality, Util.SeriesLabel(s), s.ImageCount > 0 ? s.ImageCount.ToString("N0") : "",
                        Util.Size(s.FileSize), "Waiting", q.Folder });
                    q.Item.Tag = q;
                    lock (qlock) queue.Add(q);
                    byUid[s.SeriesUID] = q;
                    lvQueue.Items.Add(q.Item);
                    added++;
                }
            }
            finally { lvQueue.EndUpdate(); }
            if (added > 0) { PumpQueue(); UiTick(); }
            return added;
        }

        void PumpQueue()
        {
            int pending;
            lock (qlock) pending = queue.Count(x => x.State == QState.Pending);
            var token = runCts.Token;
            while (activeWorkers < MaxWorkers && pending > 0)
            {
                activeWorkers++;
                pending--;
                Task.Run(() => WorkerLoop(token)).ContinueWith(t => { activeWorkers--; PumpQueue(); },
                    TaskScheduler.FromCurrentSynchronizationContext());
            }
        }

        async Task WorkerLoop(CancellationToken ct)
        {
            while (!ct.IsCancellationRequested)
            {
                QueueItem q = null;
                lock (qlock)
                {
                    foreach (var x in queue) if (x.State == QState.Pending) { q = x; break; }
                    if (q != null) q.State = QState.Active;
                }
                if (q == null) return;
                using (var itemCts = CancellationTokenSource.CreateLinkedTokenSource(ct))
                {
                    q.Cts = itemCts;
                    try { await Downloader.DownloadOne(q, itemCts.Token); }
                    catch (OperationCanceledException) { q.Status = "Canceled"; q.State = QState.Canceled; }
                    catch (Exception ex) { q.Status = "Failed: " + Util.Inner(ex); q.State = QState.Failed; }
                    finally { q.Cts = null; }
                }
            }
        }

        void UiTick()
        {
            QueueItem[] items;
            lock (qlock) items = queue.ToArray();
            double ratio = Downloader.ZipRatio;
            long total = 0;
            double done = 0;
            int nAct = 0, nPend = 0, nDone = 0, nSkip = 0, nFail = 0, nCanc = 0;
            foreach (var q in items)
            {
                long fs = Math.Max(1, q.S.FileSize);
                string st;
                Color col;
                switch (q.State)
                {
                    case QState.Pending:
                        nPend++; total += fs; st = "Waiting"; col = cText; break;
                    case QState.Active:
                        {
                            nAct++; total += fs;
                            long rec = Interlocked.Read(ref q.Received);
                            string status = q.Status;
                            double frac = status == Downloader.StUnpacking ? 0.99 : Math.Min(0.97, rec / Math.Max(1.0, q.S.FileSize * ratio));
                            done += fs * frac;
                            st = status == Downloader.StDownloading
                                ? "Downloading " + (q.S.FileSize > 0 ? (int)(frac * 100) + "%  ·  " : "") + Util.Size(rec)
                                : status;
                            col = cAccentText;
                            break;
                        }
                    case QState.Done:
                        nDone++; total += fs; done += fs; st = q.Status; col = cGreen; break;
                    case QState.Skipped:
                        nSkip++; total += fs; done += fs; st = q.Status; col = cGreen; break;
                    case QState.Failed:
                        nFail++; total += fs; done += fs; st = q.Status; col = cRed; break;
                    default:
                        nCanc++; st = "Canceled"; col = cMuted; break;
                }
                if (st != q.Shown) { q.Item.SubItems[5].Text = st; q.Shown = st; }
                if (q.Item.ForeColor != col) q.Item.ForeColor = col;

                bool final = q.State == QState.Done || q.State == QState.Skipped || q.State == QState.Failed || q.State == QState.Canceled;
                if (final && !q.Seen)
                {
                    q.Seen = true;
                    ListViewItem row;
                    if (seriesRows.TryGetValue(q.S.SeriesUID, out row) && row.ListView != null)
                    {
                        bool ok = q.State == QState.Done || q.State == QState.Skipped;
                        row.SubItems[5].Text = ok ? StSaved : q.State == QState.Failed ? "Failed" : "";
                        row.ForeColor = ok ? cGreen : q.State == QState.Failed ? cRed : cText;
                        UpdateSeriesSummary();
                    }
                }
            }

            var now = DateTime.UtcNow;
            double dt = (now - lastTick).TotalSeconds;
            long recv = Interlocked.Read(ref Downloader.TotalReceived);
            if (dt >= 0.3)
            {
                double inst = (recv - lastRecv) / dt;
                speed = speed <= 0 ? inst : speed * 0.75 + inst * 0.25;
                lastRecv = recv; lastTick = now;
            }

            bool busy = nAct + nPend > 0;
            double frac2 = total > 0 ? done / total : 0;
            bar.Value = items.Length - nCanc > 0 ? frac2 : 0;
            if (busy)
            {
                string eta = "";
                if (speed > 20 * 1024)
                {
                    string d = Util.Duration(Math.Max(0, total - done) * ratio / speed);
                    if (d != "") eta = "  ·  about " + d + " left";
                }
                lblQueueStatus.Text = (nDone + nSkip + nFail) + " of " + (items.Length - nCanc) + " series finished  ·  " +
                    (speed >= 1024 ? Util.Size((long)speed) + "/s" : "starting…") + eta;
                lblQueueStatus.ForeColor = cAccentText;
            }
            else if (items.Length > 0)
            {
                lblQueueStatus.Text = "Finished: " + nDone + " downloaded" +
                    (nSkip > 0 ? ", " + nSkip + " already on your computer" : "") +
                    (nFail > 0 ? ", " + nFail + " failed (click Retry failed)" : "") +
                    (nCanc > 0 ? ", " + nCanc + " canceled" : "");
                lblQueueStatus.ForeColor = nFail > 0 ? cRed : cGreen;
            }
            else
            {
                lblQueueStatus.Text = "Nothing queued yet. Tick the series you want above and click Download.";
                lblQueueStatus.ForeColor = cMuted;
            }

            btnCancelAll.Enabled = busy;
            btnRetry.Enabled = nFail + nCanc > 0;
            btnClear.Enabled = nDone + nSkip + nCanc + nFail > 0;

            if (taskbar != null)
            {
                try
                {
                    if (busy) { taskbar.SetProgressState(Handle, 2); taskbar.SetProgressValue(Handle, (ulong)(frac2 * 1000), 1000); }
                    else taskbar.SetProgressState(Handle, 0);
                }
                catch (Exception) { taskbar = null; }
            }
            if (busy != sleepBlocked)
            {
                Native.SetThreadExecutionState(busy ? 0x80000001u : 0x80000000u);   // keep the PC awake while downloading
                sleepBlocked = busy;
            }
            if (wasBusy && !busy)
            {
                foreach (var root in items.Select(q => q.Root).Distinct())
                {
                    string part = Path.Combine(root, ".partial");
                    try { if (Directory.Exists(part) && !Directory.EnumerateFileSystemEntries(part).Any()) Directory.Delete(part); } catch (Exception) { }
                }
                if (Form.ActiveForm != this) Native.Flash(Handle);
                SetStatus("Downloads finished. Your DICOM files are in " + saveRoot, nFail > 0);
            }
            wasBusy = busy;
        }

        void RetryItems(List<QueueItem> list)
        {
            if (list == null) lock (qlock) list = queue.ToList();
            foreach (var q in list)
                if (q.State == QState.Failed || q.State == QState.Canceled)
                {
                    q.Status = "Waiting"; q.Seen = false;
                    Interlocked.Exchange(ref q.Received, 0);
                    q.State = QState.Pending;
                }
            PumpQueue();
            UiTick();
        }

        void CancelItems(List<QueueItem> list)
        {
            foreach (var q in list)
            {
                lock (qlock) if (q.State == QState.Pending) { q.Status = "Canceled"; q.State = QState.Canceled; continue; }
                var cts = q.Cts;
                if (q.State == QState.Active && cts != null) { try { cts.Cancel(); } catch (ObjectDisposedException) { } }
            }
            UiTick();
        }

        void RemoveItems(List<QueueItem> list)
        {
            lvQueue.BeginUpdate();
            try
            {
                foreach (var q in list)
                {
                    lock (qlock)
                    {
                        if (q.State == QState.Active) continue;
                        queue.Remove(q);
                    }
                    byUid.Remove(q.S.SeriesUID);
                    lvQueue.Items.Remove(q.Item);
                }
            }
            finally { lvQueue.EndUpdate(); }
            UiTick();
        }

        void ClearFinished()
        {
            List<QueueItem> fin;
            lock (qlock) fin = queue.Where(q => q.State == QState.Done || q.State == QState.Skipped || q.State == QState.Canceled || q.State == QState.Failed).ToList();
            RemoveItems(fin);
        }

        void CancelAll(bool ask)
        {
            bool busy;
            lock (qlock) busy = queue.Any(q => q.State == QState.Active || q.State == QState.Pending);
            if (!busy) return;
            if (ask && MessageBox.Show(this, "Stop all downloads?\n\nSeries that already finished stay on your computer.", "Cancel downloads",
                MessageBoxButtons.YesNo, MessageBoxIcon.Question) != DialogResult.Yes) return;
            lock (qlock) foreach (var q in queue) if (q.State == QState.Pending) { q.Status = "Canceled"; q.State = QState.Canceled; }
            runCts.Cancel();
            runCts = new CancellationTokenSource();
            UiTick();
        }
    }

                W("manifest lookup: " + (meta.Count > 0 ? SeriesInfo.From(meta[0]).PatientID + " " + SeriesInfo.From(meta[0]).Modality : "not found"));
                var find = await Tcia.GetList("getSeries?PatientID=PANCREAS_0001", CancellationToken.None);
                W("find PANCREAS_0001: " + string.Join(", ", find.Select(r => J.S(r, "Collection")).Distinct()));
                W("downloads folder: " + Native.Downloads());
            }
            catch (Exception ex) { W("ERROR " + ex); rc = 1; }
            File.WriteAllText(Path.Combine(outDir, "selftest.log"), log.ToString());
            return rc;
        }
